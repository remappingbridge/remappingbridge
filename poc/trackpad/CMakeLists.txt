set(TRACKPAD_ADDRESS "" CACHE STRING "Optional Classic Bluetooth address of the trackpad")
if(TRACKPAD_ADDRESS AND NOT TRACKPAD_ADDRESS MATCHES "^([0-9a-fA-F][0-9a-fA-F]:)([0-9a-fA-F][0-9a-fA-F]:)([0-9a-fA-F][0-9a-fA-F]:)([0-9a-fA-F][0-9a-fA-F]:)([0-9a-fA-F][0-9a-fA-F]:)[0-9a-fA-F][0-9a-fA-F]$")
    message(FATAL_ERROR "TRACKPAD_ADDRESS must use AA:BB:CC:DD:EE:FF format")
endif()
foreach(target IN ITEMS magic_trackpad2 magic_trackpad2_macos)
    if(target STREQUAL "magic_trackpad2_macos")
        set(usb_source usb_apple.c)
    else()
        set(usb_source usb.c)
    endif()
    add_executable(${target} main.c ${usb_source} contacts.c)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${target} PRIVATE CYW43_LWIP=0 TRACKPAD_ADDRESS="${TRACKPAD_ADDRESS}")
    target_link_libraries(${target} PRIVATE pico_stdlib pico_btstack_classic
        pico_btstack_cyw43 pico_cyw43_arch_poll tinyusb_device)
    pico_enable_stdio_uart(${target} 1)
    pico_enable_stdio_usb(${target} 0)
    pico_add_extra_outputs(${target})
    if(target STREQUAL "magic_trackpad2_macos")
        target_compile_definitions(${target} PRIVATE TRACKPAD_MACOS=1)
    endif()
endforeach()
